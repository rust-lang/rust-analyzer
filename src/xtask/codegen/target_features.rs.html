1//! Extracts the list of target feature implications from rustc_target to a generated file that rust-analyzer can use for implications.
2
3use std::{collections::BTreeSet, fs, iter::repeat};

5use itertools::Itertools;
6use xshell::Shell;

8use crate::{

10    project_root,
11};

13const DESTINATION: &str = "crates/hir-ty/src/target_feature/generated.rs";
14
15/// This clones the rust repo, and so is not worth to keep up-to-date on a constant basis.
16pub(crate) fn generate(check: bool) {
17    let sh = &Shell::new().unwrap();
18    let rust_repo = clone_rust(sh);
19    let rustc_target_contents =
20        fs::read_to_string(rust_repo.join("compiler/rustc_target/src/target_features.rs")).unwrap();
21
22    let mut contents = String::from(
23        r#"
24// spellchecker:off
25
26#[expect(dead_code, reason = "stubs for rustc_target")]
27pub(super) enum Stability {
28    Stable,
29    Unstable(()),
30    InternalOnly { reason: &'static str, hard_error: bool },
31}
32use Stability::*;
33
34type ImpliedFeatures = &'static [&'static str];
35    "#,
36    );
37
38    // Identifiers of all <ARCH>_FEATURES arrays in rustc_target
39    let mut array_names = vec![];
40    // Identifiers of all Unstable(sym::<unstable feature gate>) feature gates
41    let mut unstable_feature_gates = BTreeSet::new();
42
43    // Whether or not the line iterator is currently in the middle of a feature list array
44    let mut currently_in_feature_list = false;
45    // Go through the file and do some manual parsing to keep only the lines that are part of a target feature array.
46    let target_feature_arrays = rustc_target_contents
47        .lines()
48        .filter(|line| {
49            if line.contains("&[(&str, Stability, ImpliedFeatures)]") {
50                // Start of a new feature list
51                assert!(!currently_in_feature_list);
52                currently_in_feature_list = true;
53                array_names.push(extract_list_ident_to_array_entry(line).unwrap());
54            } else if currently_in_feature_list && *line == "];" {
55                // This is the end of a feature list, so ignore all lines after this one until the start of the next feature list
56                currently_in_feature_list = false;
57                return true;
58            }
59
60            if currently_in_feature_list {
61                // Keep this line and maybe create a stub feature gate
62                unstable_feature_gates.extend(extract_to_unstable_feature_gate(line));
63                true
64            } else {
65                false
66            }
67        })
68        .interleave_shortest(repeat("\n"));
69    contents.extend(target_feature_arrays);
70
71    let mut implications_raw = String::from(
72        "pub(super) const TARGET_FEATURE_IMPLICATIONS_RAW: &[&[(&str, Stability, ImpliedFeatures)]] = &[\n",
73    );
74    implications_raw.extend(array_names);
75    implications_raw.push_str("];\n");
76    contents.push_str(&implications_raw);
77
78    let mut sym = String::from(
79        r#"
80#[expect(non_upper_case_globals, reason = "stubs for rustc_target")]
81mod sym {
82"#,
83    );
84    sym.extend(unstable_feature_gates);
85    sym.push_str("}\n// spellchecker:on\n");
86    contents.push_str(&sym);
87
88    let contents = add_preamble(crate::flags::CodegenType::TargetFeatures, reformat(contents));
89
90    let destination = project_root().join(DESTINATION);
91    ensure_file_contents(
92        crate::flags::CodegenType::TargetFeatures,
93        destination.as_path(),
94        &contents,
95        check,
96    );
97}

99/// Extract the identifier of a feature list `const $ARCH_FEATURES:` or `static $ARCH_FEATURES:` to a list entry `$ARCH_FEATURES,`
100fn extract_list_ident_to_array_entry(line: &str) -> Option<String> {
101    let ident = line.split_once(' ')?.1.split_once(':')?.0;
102    Some(format!("{ident},\n"))
103}
104
105/// Extract the identifier from `Unstable(sym::$IDENT)` and create a new mock identifier with it
106fn extract_to_unstable_feature_gate(line: &str) -> Option<String> {
107    let ident = line.split_once("Unstable(sym::")?.1.split_once(')')?.0;
108
109    Some(format!("pub(in super::super) const {ident}: () = ();\n"))
110}
