1//! A temporary named file that will be deleted on drop, and on operating systems that support that,
2//! also when the process exits (including being killed).
3
4use std::{
5    fs::File,
6    io,
7    path::{Path, PathBuf},
8};
9
10pub struct NamedTempFile {
11    _file: Option<File>,
12    path: PathBuf,
13    delete_on_drop: bool,
14}
15
16impl NamedTempFile {
17    pub fn new(prefix: &str) -> io::Result<NamedTempFile> {
18        imp::create(prefix)
19    }
20
21    /// Creates a new `NamedTempFile` that is a copy of an existing file.
22    pub fn new_from_existing(prefix: &str, existing: &Path) -> io::Result<NamedTempFile> {
23        let result = NamedTempFile::new(prefix)?;
24        std::fs::copy(existing, &result.path)?;
25        Ok(result)
26    }
27
28    /// Creates a `NamedTempFile` from a path, without deleting it on drop.
29    #[inline]
30    pub fn from_path(path: PathBuf) -> NamedTempFile {
31        NamedTempFile { _file: None, path, delete_on_drop: false }
32    }
33
34    #[inline]
35    pub fn path(&self) -> &Path {
36        &self.path
37    }
38}
39
40impl Drop for NamedTempFile {
41    fn drop(&mut self) {
42        if self.delete_on_drop && std::fs::remove_file(&self.path).is_err() {
43            tracing::info!("cannot remove temporary file {}", self.path.display());
44        }
45    }
46}

48pub struct NamedTempDir {
49    path: PathBuf,
50}
51
52impl NamedTempDir {
53    pub fn new(prefix: &str) -> io::Result<NamedTempDir> {
54        general_imp::create(prefix, |_options, path| std::fs::create_dir(path))
55            .map(|((), path)| NamedTempDir { path })
56    }
57
58    #[inline]
59    pub fn path(&self) -> &Path {
60        &self.path
61    }
62}
63
64impl Drop for NamedTempDir {
65    fn drop(&mut self) {
66        if std::fs::remove_dir_all(&self.path).is_err() {
67            tracing::info!("cannot remove temporary directory {}", self.path.display());
68        }
69    }
70}

72mod general_imp {
73    use std::{
74        fs::OpenOptions,
75        io::{self, ErrorKind},
76        path::{Path, PathBuf},
77        sync::atomic::{AtomicU32, Ordering},
78    };
79
80    static INTERNAL_COUNTER: AtomicU32 = AtomicU32::new(0);
81
82    pub(super) fn create<T>(
83        prefix: &str,
84        mut create: impl FnMut(OpenOptions, &Path) -> io::Result<T>,
85    ) -> io::Result<(T, PathBuf)> {
86        let temp_dir = std::env::temp_dir().canonicalize()?;
87        let pid = std::process::id();
88        loop {
89            let path = temp_dir.join(format!(
90                "{prefix}{pid:x}-{:x}",
91                INTERNAL_COUNTER.fetch_add(1, Ordering::AcqRel),
92            ));
93            let mut open_options = OpenOptions::new();
94            open_options.create_new(true);
95            match create(open_options, &path) {
96                Err(e) if e.kind() == ErrorKind::AlreadyExists => {}
97                Err(e) => {
98                    return Err(io::Error::new(
99                        e.kind(),
100                        format!("error creating directory {path:?}: {e}"),
101                    ));
102                }
103                Ok(file) => {
104                    return Ok((file, path));
105                }
106            }
107        }
108    }
109}

111#[cfg(any(
112    target_os = "linux",
113    target_os = "freebsd",
114    target_os = "openbsd",
115    target_os = "netbsd",
116))]
117mod imp {
118    use std::{
119        ffi::CString,
120        io,
121        os::{
122            fd::{AsRawFd, RawFd},
123            unix::ffi::OsStrExt,
124        },
125    };
126
127    use super::*;
128
129    #[cfg(target_os = "linux")]
130    fn path_after_unlink(fd: RawFd) -> PathBuf {
131        PathBuf::from(format!("/proc/self/fd/{fd}"))
132    }
133
134    #[cfg(any(target_os = "freebsd", target_os = "openbsd", target_os = "netbsd"))]
135    fn path_after_unlink(fd: RawFd) -> PathBuf {
136        PathBuf::from(format!("/dev/fd/{fd}"))
137    }
138
139    pub(super) fn create(prefix: &str) -> io::Result<NamedTempFile> {
140        let (file, mut path) = general_imp::create(prefix, |options, path| options.open(path))?;
141        let mut delete_on_drop = true;
142        if let Ok(original_path) = CString::new(path.as_os_str().as_bytes()) {
143            // Unlinking the file will *not* remove it per the POSIX specification since it is open.
144            // We cannot use `std::fs::remove_file()`, since, while currently using `unlink()`, it does
145            // not guarantee it will use it.
146            if unsafe { libc::unlink(original_path.as_ptr()) } == 0 {
147                path = path_after_unlink(file.as_raw_fd());
148                delete_on_drop = false;
149            }
150        }
151        Ok(NamedTempFile { _file: Some(file), path, delete_on_drop })
152    }
153}

155#[cfg(windows)]
156mod imp {
157    use std::os::windows::fs::OpenOptionsExt;
158
159    use super::*;
160
161    const FILE_ATTRIBUTE_TEMPORARY: u32 = 0x100;
162    const FILE_FLAG_DELETE_ON_CLOSE: u32 = 0x04000000;
163
164    pub(super) fn create(prefix: &str) -> io::Result<NamedTempFile> {
165        let (file, path) = general_imp::create(prefix, |mut options, path| {
166            options
167                .attributes(FILE_ATTRIBUTE_TEMPORARY)
168                .custom_flags(FILE_FLAG_DELETE_ON_CLOSE)
169                .open(path)
170        })?;
171        Ok(NamedTempFile { _file: Some(file), path, delete_on_drop: false })
172    }
173}
