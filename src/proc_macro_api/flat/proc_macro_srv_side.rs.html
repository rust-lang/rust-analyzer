1//! Conversion from/to the flat tree for the proc macro server.
2
3use crate::{
4    flat::{FlatTree, ReaderTrait, SpanTransformer, SubtreeOrLeafRef, WriterTrait},
5    token_stream::{Group, SpanLike, TokenStream, TokenTree},
6};
7
8struct Writer;
9
10impl<'a, Span: Copy + 'a> WriterTrait<'a, Span> for Writer {
11    type Subtree = &'a Group<Span>;
12    type Leaf = &'a tt::Leaf<Span>;
13
14    type SubtreeIter = Option<std::slice::Iter<'a, TokenTree<Span>>>;
15
16    fn subtree_data(subtree: Self::Subtree) -> (usize, tt::Delimiter<Span>, Self::SubtreeIter) {
17        (subtree.stream_len(), subtree.delimiter, subtree.stream.as_ref().map(|it| it.iter()))
18    }
19
20    fn subtree_iter_next(iter: &mut Self::SubtreeIter) -> Option<SubtreeOrLeafRef<'a, Span, Self>> {
21        iter.as_mut()?.next().map(|item| match item {
22            TokenTree::Leaf(leaf) => SubtreeOrLeafRef::Leaf(leaf),
23            TokenTree::Group(group) => SubtreeOrLeafRef::Subtree(group),
24        })
25    }
26}
27
28struct Reader;
29
30impl<Span: SpanLike> ReaderTrait<Span> for Reader {
31    type TokenTree = TokenTree<Span>;
32
33    fn leaf(leaf: tt::Leaf<Span>) -> Self::TokenTree {
34        TokenTree::Leaf(leaf)
35    }
36
37    fn append_subtree(
38        delimiter: tt::Delimiter<Span>,
39        children: Vec<Self::TokenTree>,
40        insert_into: &mut Vec<Self::TokenTree>,
41    ) {
42        insert_into.push(TokenTree::Group(Group {
43            delimiter,
44            stream: TokenStream::new_or_empty(children),
45        }));
46    }
47}
48
49impl FlatTree {
50    pub fn from_tokenstream<ST: SpanTransformer>(
51        tokenstream: TokenStream<ST::Span>,
52        call_site: ST::Span,
53        version: u32,
54        span_data_table: &mut ST::Table,
55    ) -> FlatTree {
56        let root = if let Some(group) = tokenstream.as_single_group() {
57            group.clone()
58        } else {
59            Group {
60                delimiter: tt::Delimiter {
61                    open: call_site,
62                    close: call_site,
63                    kind: tt::DelimiterKind::Invisible,
64                },
65                stream: Some(tokenstream),
66            }
67        };
68        FlatTree::to_flat_tree::<ST, Writer>(&root, version, span_data_table)
69    }
70
71    pub fn to_tokenstream<ST: SpanTransformer<Span: SpanLike>>(
72        self,
73        version: u32,
74        span_data_table: &ST::Table,
75    ) -> TokenStream<ST::Span> {
76        let (top_delimiter, top_children) =
77            self.from_flat_tree::<ST, Reader>(version, span_data_table);
78        let result = if top_delimiter.kind == tt::DelimiterKind::Invisible {
79            top_children
80        } else {
81            vec![TokenTree::Group(Group {
82                delimiter: top_delimiter,
83                stream: TokenStream::new_or_empty(top_children),
84            })]
85        };
86        TokenStream::new(result)
87    }
88}
