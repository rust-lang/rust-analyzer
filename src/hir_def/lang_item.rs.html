39/// Salsa query. This will look for lang items in a specific crate.
40#[salsa::tracked(returns(as_deref))]
41pub fn crate_lang_items(db: &dyn SourceDatabase, krate: Crate) -> Option<Box<LangItems>> {
42    if !UnstableFeatures::query(db, krate).lang_items {
43        // First query this, *before* querying the def map, to not create a dependency edge.
44        return None;
45    }
46
47    let mut lang_items = LangItems::default();
48
49    let _p = tracing::info_span!("crate_lang_items_query", ?krate).entered();
50
51    let crate_def_map = crate_def_map(db, krate);
52    for (_, module_data) in crate_def_map.modules() {
53        for impl_def in module_data.scope.inherent_impls() {
54            lang_items.collect_lang_item(db, impl_def);
55            for &(_, assoc) in impl_def.impl_items(db).items.iter() {
56                match assoc {
57                    AssocItemId::FunctionId(f) => lang_items.collect_lang_item(db, f),
58                    AssocItemId::TypeAliasId(t) => lang_items.collect_lang_item(db, t),
59                    AssocItemId::ConstId(c) => lang_items.collect_lang_item(db, c),
60                }
61            }
62        }
63
64        for def in module_data.scope.declarations() {
65            match def {
66                ModuleDefId::TraitId(trait_) => {
67                    lang_items.collect_lang_item(db, trait_);
68                    TraitItems::query(db, trait_).items.iter().for_each(|&(_, assoc_id)| {
69                        match assoc_id {
70                            AssocItemId::FunctionId(f) => {
71                                lang_items.collect_lang_item(db, f);
72                            }
73                            AssocItemId::TypeAliasId(alias) => {
74                                lang_items.collect_lang_item(db, alias)
75                            }
76                            AssocItemId::ConstId(c) => lang_items.collect_lang_item(db, c),
77                        }
78                    });
79                }
80                ModuleDefId::AdtId(AdtId::EnumId(e)) => {
81                    lang_items.collect_lang_item(db, e);
82                    e.enum_variants(db).variants.values().for_each(|&(id, _)| {
83                        lang_items.collect_lang_item(db, id);
84                    });
85                }
86                ModuleDefId::AdtId(AdtId::StructId(s)) => {
87                    lang_items.collect_lang_item(db, s);
88                }
89                ModuleDefId::AdtId(AdtId::UnionId(u)) => {
90                    lang_items.collect_lang_item(db, u);
91                }
92                ModuleDefId::FunctionId(f) => {
93                    lang_items.collect_lang_item(db, f);
94                }
95                ModuleDefId::StaticId(s) => {
96                    lang_items.collect_lang_item(db, s);
97                }
98                ModuleDefId::TypeAliasId(t) => {
99                    lang_items.collect_lang_item(db, t);
100                }
101                ModuleDefId::ConstId(c) => lang_items.collect_lang_item(db, c),
102                _ => {}
103            }
104        }
105    }
106
107    if matches!(krate.data(db).origin, base_db::CrateOrigin::Lang(base_db::LangCrateOrigin::Core)) {
108        lang_items.fill_non_lang_core_items(db, crate_def_map);
109    }
110
111    lang_items.resolve_manually(db);
112
113    if lang_items.is_empty() {
114        None
115    } else {
116        lang_items.shrink_to_fit();
117        Some(Box::new(lang_items))
118    }
119}

121// A firewall query, so that changing crate code for non-`feature(lang_items)` crate will not require recomputing
122// the more expensive `lang_items()` query.
123#[salsa::tracked(returns(deref))]
124fn crate_extern_prelude(db: &dyn SourceDatabase, start_crate: Crate) -> Box<[Crate]> {
125    crate_local_def_map(db, start_crate)
126        .local(db)
127        .extern_prelude()
128        .map(|(_, (module, _))| module.krate(db))
129        .filter(|&krate| {
130            // Some crates declares themselves as extern crate like `extern crate self as core`.
131            // Ignore these to prevent cycles.
132            krate != start_crate
133        })
134        .collect()
135}

137/// Salsa query. Look for a lang items, starting from the specified crate and recursively
138/// traversing its dependencies.
139#[salsa::tracked(returns(ref))]
140pub fn lang_items(db: &dyn SourceDatabase, start_crate: Crate) -> LangItems {
141    let _p = tracing::debug_span!("lang_items_query", ?start_crate).entered();
142
143    let mut result = crate_lang_items(db, start_crate).cloned().unwrap_or_default();
144
145    // Our `CrateGraph` eagerly inserts sysroot dependencies like `core` or `std` into dependencies
146    // even if the target crate has `#![no_std]`, `#![no_core]` or shadowed sysroot dependencies
147    // like `dependencies.std.path = ".."`. So we use `extern_prelude()` instead of
148    // `CrateData.dependencies` here, which has already come through such sysroot complexities
149    // while nameres.
150    //
151    // See https://github.com/rust-lang/rust-analyzer/pull/20475 for details.
152    for &krate in crate_extern_prelude(db, start_crate) {
153        result.merge_prefer_other(lang_items(db, krate));
154    }
155
156    result
157}

159impl LangItems {
160    fn collect_lang_item<T>(&mut self, db: &dyn SourceDatabase, item: T)
161    where
162        T: Into<AttrDefId> + Into<LangItemTarget> + Copy,
163    {
164        if let Some(lang_item) = AttrFlags::lang_item(db, item.into()) {
165            tracing::trace!("found lang item `{lang_item}`");
166            self.assign_lang_item(lang_item, item.into());
167        }
168    }
169}

171fn resolve_core_trait(
172    db: &dyn SourceDatabase,
173    core_def_map: &DefMap,
174    modules: &[Symbol],
175    name: Symbol,
176) -> Option<TraitId> {
177    let mut current = &core_def_map[core_def_map.root];
178    for module in modules {
179        let Some((ModuleDefId::ModuleId(cur), _)) =
180            current.scope.type_(&Name::new_symbol_root(module.clone()))
181        else {
182            return None;
183        };
184        if cur.krate(db) != core_def_map.krate() || cur.block(db) != core_def_map.block_id() {
185            return None;
186        }
187        current = &core_def_map[cur];
188    }
189    let Some((ModuleDefId::TraitId(trait_), _)) = current.scope.type_(&Name::new_symbol_root(name))
190    else {
191        return None;
192    };
193    Some(trait_)
194}

196fn resolve_core_macro(
197    db: &dyn SourceDatabase,
198    core_def_map: &DefMap,
199    modules: &[Symbol],
200    name: Symbol,
201) -> Option<MacroId> {
202    let mut current = &core_def_map[core_def_map.root];
203    for module in modules {
204        let Some((ModuleDefId::ModuleId(cur), _)) =
205            current.scope.type_(&Name::new_symbol_root(module.clone()))
206        else {
207            return None;
208        };
209        if cur.krate(db) != core_def_map.krate() || cur.block(db) != core_def_map.block_id() {
210            return None;
211        }
212        current = &core_def_map[cur];
213    }
214    current.scope.makro(&Name::new_symbol_root(name))
215}

217impl LangItems {
218    fn resolve_manually(&mut self, db: &dyn SourceDatabase) {
219        let parent_trait =
220            |lang_item: &mut Option<TraitId>, def: Option<FunctionId>| match def?.loc(db).container
221            {
222                ItemContainerId::TraitId(trait_) => {
223                    *lang_item = Some(trait_);
224                    Some(trait_)
225                }
226                _ => None,
227            };
228        let assoc_types =
229            |trait_: TraitId, assoc_types: &mut [(&mut Option<TypeAliasId>, Symbol)]| {
230                let trait_items = trait_.trait_items(db);
231                for (assoc_type, name) in assoc_types {
232                    **assoc_type =
233                        trait_items.associated_type_by_name(&Name::new_symbol_root(name.clone()));
234                }
235            };
236        let methods = |trait_: TraitId, assoc_types: &mut [(&mut Option<FunctionId>, Symbol)]| {
237            let trait_items = trait_.trait_items(db);
238            for (assoc_type, name) in assoc_types {
239                **assoc_type = trait_items.method_by_name(&Name::new_symbol_root(name.clone()));
240            }
241        };
242        (|| {
243            let into_future = parent_trait(&mut self.IntoFuture, self.IntoFutureIntoFuture)?;
244            assoc_types(into_future, &mut [(&mut self.IntoFutureOutput, sym::Output)]);
245            Some(())
246        })();
247
248        (|| {
249            let into_iterator = parent_trait(&mut self.IntoIterator, self.IntoIterIntoIter)?;
250            assoc_types(
251                into_iterator,
252                &mut [
253                    (&mut self.IntoIteratorItem, sym::Item),
254                    (&mut self.IntoIterIntoIterType, sym::IntoIter),
255                ],
256            );
257            Some(())
258        })();
259
260        (|| {
261            assoc_types(self.Iterator?, &mut [(&mut self.IteratorItem, sym::Item)]);
262            Some(())
263        })();
264
265        (|| {
266            assoc_types(self.AsyncIterator?, &mut [(&mut self.AsyncIteratorItem, sym::Item)]);
267            Some(())
268        })();
269
270        for (op_trait, op_method, op_method_name) in [
271            (self.Fn, &mut self.Fn_call, sym::call),
272            (self.FnMut, &mut self.FnMut_call_mut, sym::call_mut),
273            (self.FnOnce, &mut self.FnOnce_call_once, sym::call_once),
274            (self.AsyncFn, &mut self.AsyncFn_async_call, sym::async_call),
275            (self.AsyncFnMut, &mut self.AsyncFnMut_async_call_mut, sym::async_call_mut),
276            (self.AsyncFnOnce, &mut self.AsyncFnOnce_async_call_once, sym::async_call_once),
277            (self.Not, &mut self.Not_not, sym::not),
278            (self.Neg, &mut self.Neg_neg, sym::neg),
279            (self.Add, &mut self.Add_add, sym::add),
280            (self.Mul, &mut self.Mul_mul, sym::mul),
281            (self.Sub, &mut self.Sub_sub, sym::sub),
282            (self.Div, &mut self.Div_div, sym::div),
283            (self.Rem, &mut self.Rem_rem, sym::rem),
284            (self.Shl, &mut self.Shl_shl, sym::shl),
285            (self.Shr, &mut self.Shr_shr, sym::shr),
286            (self.BitXor, &mut self.BitXor_bitxor, sym::bitxor),
287            (self.BitOr, &mut self.BitOr_bitor, sym::bitor),
288            (self.BitAnd, &mut self.BitAnd_bitand, sym::bitand),
289            (self.AddAssign, &mut self.AddAssign_add_assign, sym::add_assign),
290            (self.MulAssign, &mut self.MulAssign_mul_assign, sym::mul_assign),
291            (self.SubAssign, &mut self.SubAssign_sub_assign, sym::sub_assign),
292            (self.DivAssign, &mut self.DivAssign_div_assign, sym::div_assign),
293            (self.RemAssign, &mut self.RemAssign_rem_assign, sym::rem_assign),
294            (self.ShlAssign, &mut self.ShlAssign_shl_assign, sym::shl_assign),
295            (self.ShrAssign, &mut self.ShrAssign_shr_assign, sym::shr_assign),
296            (self.BitXorAssign, &mut self.BitXorAssign_bitxor_assign, sym::bitxor_assign),
297            (self.BitOrAssign, &mut self.BitOrAssign_bitor_assign, sym::bitor_assign),
298            (self.BitAndAssign, &mut self.BitAndAssign_bitand_assign, sym::bitand_assign),
299            (self.Debug, &mut self.Debug_fmt, sym::fmt),
300            (self.Deref, &mut self.Deref_deref, sym::deref),
301            (self.DerefMut, &mut self.DerefMut_deref_mut, sym::deref_mut),
302            (self.Index, &mut self.Index_index, sym::index),
303            (self.IndexMut, &mut self.IndexMut_index_mut, sym::index_mut),
304        ] {
305            (|| {
306                methods(op_trait?, &mut [(op_method, op_method_name)]);
307                Some(())
308            })();
309        }
310        (|| {
311            methods(
312                self.PartialEq?,
313                &mut [(&mut self.PartialEq_eq, sym::eq), (&mut self.PartialEq_ne, sym::ne)],
314            );
315            Some(())
316        })();
317        (|| {
318            methods(
319                self.PartialOrd?,
320                &mut [
321                    (&mut self.PartialOrd_le, sym::le),
322                    (&mut self.PartialOrd_lt, sym::lt),
323                    (&mut self.PartialOrd_ge, sym::ge),
324                    (&mut self.PartialOrd_gt, sym::gt),
325                ],
326            );
327            Some(())
328        })();
329        (|| {
330            methods(
331                self.Drop?,
332                &mut [(&mut self.Drop_drop, sym::drop), (&mut self.Drop_pin_drop, sym::pin_drop)],
333            );
334            Some(())
335        })();
336    }

398#[salsa::tracked(returns(as_deref))]
399pub fn crate_notable_traits(db: &dyn SourceDatabase, krate: Crate) -> Option<Box<[TraitId]>> {
400    let mut traits = Vec::new();
401
402    let crate_def_map = crate_def_map(db, krate);
403
404    for (_, module_data) in crate_def_map.modules() {
405        for def in module_data.scope.declarations() {
406            if let ModuleDefId::TraitId(trait_) = def
407                && AttrFlags::query(db, trait_.into()).contains(AttrFlags::IS_DOC_NOTABLE_TRAIT)
408            {
409                traits.push(trait_);
410            }
411        }
412    }
413
414    if traits.is_empty() { None } else { Some(traits.into_iter().collect()) }
415}

417macro_rules! assign_solver {
418    ( $self:ident, $def:ident, $lang_item:ident, TraitId, solver ) => {
419        $self.assign_solver_trait(SolverTraitLangItem::$lang_item, $def);
420    };
421    ( $self:ident, $def:ident, $lang_item:ident, TypeAliasId, solver ) => {
422        $self.assign_solver_projection(SolverProjectionLangItem::$lang_item, $def);
423    };
424    ( $self:ident, $def:ident, $lang_item:ident, StructId, solver ) => {
425        $self.assign_solver_adt(SolverAdtLangItem::$lang_item, $def);
426    };
427    ( $self:ident, $def:ident, $lang_item:ident, EnumId, solver ) => {
428        $self.assign_solver_adt(SolverAdtLangItem::$lang_item, $def);
429    };
430    ( $self:ident, $def:ident, $lang_item:ident, UnionId, solver ) => {
431        $self.assign_solver_adt(SolverAdtLangItem::$lang_item, $def);
432    };
433    ( $self:ident, $def:ident, $lang_item:ident, $ty:ident ) => {};
434}

436macro_rules! solver_exhaustiveness_check {
437    (
438        $dollar:tt, $solver_ty:ident, $id_ty:ident,
439        [ $($id_tys:ident),* ]
440
441        $( $lang_item:ident, $target:ident $(, $solver:ident)?; )*
442    ) => {
443        const _: () = {
444            #[expect(unreachable_patterns, dead_code)]
445            fn exhaustiveness_check(lang_item: $solver_ty) {
446                macro_rules! one_variant {
447                    $(
448                        (
449                            $dollar lang_item:ident, $id_tys, solver
450                        ) => {
451                            ($solver_ty::$dollar lang_item, true)
452                        };
453                    )*
454                    ( $dollar($t:tt)* ) => {
455                        (_, false)
456                    };
457                }
458
459                match (lang_item, true) {
460                    $(
461                        one_variant!( $lang_item, $target $(, $solver )? ) => {}
462                    )*
463                }
464            }
465        };
466    };
467}

469macro_rules! get_solver_count {
470    (
471        $dollar:tt, $solver_ty:ident,
472        [ $($id_tys:ident),* ]
473
474        $( $lang_item:ident, $target:ident $(, $solver:ident)?; )*
475    ) => {{
476        let mut max_discriminant = 0;
477        macro_rules! one_variant {
478            $(
479                (
480                    $dollar lang_item:ident, $id_tys, solver
481                ) => {
482                    if $solver_ty::$dollar lang_item as usize > max_discriminant {
483                        max_discriminant = $solver_ty::$dollar lang_item as usize;
484                    }
485                };
486            )*
487            ( $dollar($t:tt)* ) => {};
488        }
489        $(
490            one_variant!($lang_item, $target $(, $solver)?);
491        )*
492        max_discriminant + 1
493    }};
494}

496fn merge_array_prefer_other<T: Copy, const N: usize>(
497    this: &mut [Option<T>; N],
498    other: &[Option<T>; N],
499) {
500    for (this, other) in std::iter::zip(this, other) {
501        *this = other.or(*this);
502    }
503}

505macro_rules! language_item_table {
506    (
507        $dollar:tt
508        $LangItems:ident =>
509        $( $lang_item:ident, $module:ident :: $name:ident, $target:ident $(, $solver:ident)?; )*
510
511        @non_lang_core_traits:
512
513        $( core::$($non_lang_trait_module:ident)::*, $non_lang_trait:ident; )*
514
515        @non_lang_core_macros:
516
517        $( core::$($non_lang_macro_module:ident)::*, $non_lang_macro:ident, $non_lang_macro_field:ident; )*
518
519        @resolve_manually:
520
521        $( $resolve_manually:ident, $resolve_manually_type:ident; )*
522    ) => {
523        const SOLVER_TRAITS_LEN: usize = get_solver_count! {
524            $dollar, SolverTraitLangItem,
525            [ TraitId ]
526
527            $( $lang_item, $target $(, $solver)?; )*
528        };
529        const SOLVER_PROJECTIONS_LEN: usize = get_solver_count! {
530            $dollar, SolverProjectionLangItem,
531            [ TypeAliasId ]
532
533            $( $lang_item, $target $(, $solver)?; )*
534        };
535        const SOLVER_ADTS_LEN: usize = get_solver_count! {
536            $dollar, SolverAdtLangItem,
537            [ StructId, EnumId, UnionId ]
538
539            $( $lang_item, $target $(, $solver)?; )*
540        };
541
542        #[allow(non_snake_case)] // FIXME: Should we remove this?
543        #[derive(Default, Clone)]
544        pub struct $LangItems {
545            $(
546                pub $lang_item: Option<$target>,
547            )*
548            $(
549                pub $non_lang_trait: Option<TraitId>,
550            )*
551            $(
552                pub $non_lang_macro_field: Option<MacroId>,
553            )*
554            $(
555                pub $resolve_manually: Option<$resolve_manually_type>,
556            )*
557
558            solver_trait: [Option<TraitId>; SOLVER_TRAITS_LEN],
559            solver_adt: [Option<AdtId>; SOLVER_ADTS_LEN],
560            solver_projection: [Option<TypeAliasId>; SOLVER_PROJECTIONS_LEN],
561
562            reverse_solver_trait: FxHashMap<TraitId, SolverTraitLangItem>,
563            reverse_solver_adt: FxHashMap<AdtId, SolverAdtLangItem>,
564            reverse_solver_projection: FxHashMap<TypeAliasId, SolverProjectionLangItem>,
565        }
566
567        impl std::fmt::Debug for $LangItems {
568            fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
569                let mut debug = f.debug_struct(stringify!($LangItems));
570                let Self {
571                    $( $lang_item, )*
572                    $( $non_lang_trait, )*
573                    $( $non_lang_macro_field, )*
574                    $( $resolve_manually, )*
575
576                    solver_trait: _,
577                    solver_adt: _,
578                    solver_projection: _,
579                    reverse_solver_trait: _,
580                    reverse_solver_adt: _,
581                    reverse_solver_projection: _,
582                } = self;
583                $(
584                    if let Some(it) = $lang_item {
585                        debug.field(stringify!($lang_item), it);
586                    }
587                )*
588                $(
