50pub struct ScopeData {
51    parent: Option<ScopeId>,

56#[derive(Debug, PartialEq, Eq)]
57enum ScopeKind {
58    None,
59    Block { id: BlockId, label: Option<LabelId> },
60    Label(LabelId),
61    MacroDef(Box<MacroDefId>),
62}

67    pub fn body_expr_scopes(db: &dyn SourceDatabase, def: DefWithBodyId) -> ExprScopes {
68        let body = Body::of(db, def);
69        let mut scopes = ExprScopes::new_body(body);

78        let mut scopes = ExprScopes::new_store(store, roots);

85        let fields = VariantFields::of(db, def);

93impl ExprScopes {
94    #[inline]
95    pub fn of(db: &dyn SourceDatabase, def: impl Into<ExpressionStoreOwnerId>) -> &ExprScopes {
96        match def.into() {
97            ExpressionStoreOwnerId::Body(def) => Self::body_expr_scopes(db, def),
98            ExpressionStoreOwnerId::Signature(def) => Self::sig_expr_scopes(db, def),
99            ExpressionStoreOwnerId::VariantFields(variant_id) => {
100                Self::variant_scopes(db, variant_id)
101            }
102        }
103    }
104
105    pub fn entries(&self, scope: ScopeId) -> &[ScopeEntry] {
106        &self.scope_entries[self.scopes[scope].entries.clone()]
107    }
108
109    /// If `scope` refers to a block expression scope, returns the corresponding `BlockId`.
110    pub fn block(&self, scope: ScopeId) -> Option<BlockId> {
111        match self.scopes[scope].kind {
112            ScopeKind::Block { id, label: _ } => Some(id),
113            ScopeKind::None | ScopeKind::Label(_) | ScopeKind::MacroDef(_) => None,
114        }
115    }
116
117    /// If `scope` refers to a macro def scope, returns the corresponding `MacroId`.
118    #[allow(clippy::borrowed_box)] // If we return `&MacroDefId` we need to move it, this way we just clone the `Box`.
119    pub fn macro_def(&self, scope: ScopeId) -> Option<&Box<MacroDefId>> {
120        match &self.scopes[scope].kind {
121            ScopeKind::MacroDef(macro_def) => Some(macro_def),
122            ScopeKind::None | ScopeKind::Block { id: _, label: _ } | ScopeKind::Label(_) => None,
123        }
124    }
125
126    /// If `scope` refers to a labeled expression scope, returns the corresponding `Label`.
127    pub fn label(&self, scope: ScopeId) -> Option<LabelId> {
128        match &self.scopes[scope].kind {
129            &ScopeKind::Block { id: _, label } => label,
130            &ScopeKind::Label(label) => Some(label),
131            ScopeKind::None | ScopeKind::MacroDef(_) => None,
132        }
133    }
134
135    /// Returns the scopes in ascending order.
136    pub fn scope_chain(&self, scope: Option<ScopeId>) -> impl Iterator<Item = ScopeId> + '_ {
137        std::iter::successors(scope, move |&scope| self.scopes[scope].parent)
138    }
139
140    pub fn resolve_name_in_scope(&self, scope: ScopeId, name: &Name) -> Option<&ScopeEntry> {
141        self.scope_chain(Some(scope))
142            .find_map(|scope| self.entries(scope).iter().find(|it| it.name == *name))
143    }
144
145    pub fn scope_for(&self, expr: ExprId) -> Option<ScopeId> {
146        self.scope_by_expr.get(expr).copied()
147    }
148
149    pub fn scope_by_expr(&self) -> &ArenaMap<ExprId, ScopeId> {
150        &self.scope_by_expr
151    }
152}

154fn empty_entries(idx: usize) -> IdxRange<ScopeEntry> {
155    IdxRange::new(Idx::from_raw(RawIdx::from(idx as u32))..Idx::from_raw(RawIdx::from(idx as u32)))
156}

158impl ExprScopes {
159    fn new_body(body: &Body) -> ExprScopes {
160        let mut scopes = ExprScopes {
161            scopes: Arena::default(),
162            scope_entries: Arena::default(),
163            scope_by_expr: ArenaMap::with_capacity(
164                body.expr_only.as_ref().map_or(0, |it| it.exprs.len()),
165            ),
166        };
167        let root = scopes.root_scope();
168        if let Some(Param { formal: self_param, user_written: _ }) = body.self_param {
169            scopes.add_bindings(body, root, self_param, body.binding_hygiene(self_param));
170        }
171        let mut visitor =
172            ExprScopeVisitor { store: body, scopes: &mut scopes, scope: root, const_scope: root };
173        body.params.iter().for_each(|param| visitor.on_pat(param.formal));
174        visitor.on_expr(body.root_expr());
175        scopes
176    }
177
178    fn new_store(store: &ExpressionStore, roots: impl IntoIterator<Item = ExprId>) -> ExprScopes {
179        let mut scopes = ExprScopes {
180            scopes: Arena::default(),
181            scope_entries: Arena::default(),
182            scope_by_expr: ArenaMap::with_capacity(
183                store.expr_only.as_ref().map_or(0, |it| it.exprs.len()),
184            ),
185        };
186        let root = scopes.root_scope();
187        for root_expr in roots {
188            let scope = scopes.new_scope(root);
189            ExprScopeVisitor { store, scopes: &mut scopes, scope, const_scope: scope }
190                .on_expr(root_expr);
191        }
192        scopes
193    }
194
195    fn root_scope(&mut self) -> ScopeId {
196        self.scopes.alloc(ScopeData {
197            parent: None,
198            kind: ScopeKind::None,
199            entries: empty_entries(self.scope_entries.len()),
200        })
201    }
202
203    fn new_scope(&mut self, parent: ScopeId) -> ScopeId {
204        self.scopes.alloc(ScopeData {
205            parent: Some(parent),
206            kind: ScopeKind::None,
207            entries: empty_entries(self.scope_entries.len()),
208        })
209    }
210
211    fn new_labeled_scope(&mut self, parent: ScopeId, label: Option<LabelId>) -> ScopeId {
212        let kind = match label {
213            Some(label) => ScopeKind::Label(label),
214            None => ScopeKind::None,
215        };
216        self.scopes.alloc(ScopeData {
217            parent: Some(parent),
218            kind,
219            entries: empty_entries(self.scope_entries.len()),
220        })
221    }
222
223    fn new_block_scope(
224        &mut self,
225        parent: ScopeId,
226        block: Option<BlockId>,
227        label: Option<LabelId>,
228    ) -> ScopeId {
229        let kind = match (block, label) {
230            (Some(id), label) => ScopeKind::Block { id, label },
231            (None, Some(label)) => ScopeKind::Label(label),
232            (None, None) => ScopeKind::None,
233        };
234        self.scopes.alloc(ScopeData {
235            parent: Some(parent),
236            kind,
237            entries: empty_entries(self.scope_entries.len()),
238        })
239    }
240
241    fn new_macro_def_scope(&mut self, parent: ScopeId, macro_id: Box<MacroDefId>) -> ScopeId {
242        self.scopes.alloc(ScopeData {
243            parent: Some(parent),
244            kind: ScopeKind::MacroDef(macro_id),
245            entries: empty_entries(self.scope_entries.len()),
246        })
247    }
248
249    fn add_bindings(
250        &mut self,
251        store: &ExpressionStore,
252        scope: ScopeId,
253        binding: BindingId,
254        hygiene: HygieneId,
255    ) {
256        let Binding { name, .. } = &store[binding];
257        let entry = self.scope_entries.alloc(ScopeEntry { name: name.clone(), binding, hygiene });
258        self.scopes[scope].entries =
259            IdxRange::new_inclusive(self.scopes[scope].entries.start()..=entry);
260    }
261
262    fn set_scope(&mut self, node: ExprId, scope: ScopeId) {
263        self.scope_by_expr.insert(node, scope);
264    }
265
266    fn shrink_to_fit(&mut self) {
267        let ExprScopes { scopes, scope_entries, scope_by_expr } = self;
268        scopes.shrink_to_fit();
269        scope_entries.shrink_to_fit();
270        scope_by_expr.shrink_to_fit();
271    }
272}

274struct ExprScopeVisitor<'a> {
275    store: &'a ExpressionStore,
276    scopes: &'a mut ExprScopes,
277    scope: ScopeId,
278    const_scope: ScopeId,
279}
280
281impl ExprScopeVisitor<'_> {
282    fn with_scope(&mut self, scope: ScopeId, f: impl FnOnce(&mut Self)) {
283        let old_scope = mem::replace(&mut self.scope, scope);
284        f(self);
285        self.scope = old_scope;
286    }
287
288    fn visit_block(
289        &mut self,
290        expr: ExprId,
291        id: Option<BlockId>,
292        statements: &[Statement],
293        tail: Option<ExprId>,
294        label: Option<LabelId>,
295    ) {
296        let scope = self.scopes.new_block_scope(self.scope, id, label);
297        self.with_scope(scope, |this| {
298            let old_const_scope = if id.is_some() {
299                let const_scope = this.scopes.new_block_scope(this.const_scope, id, None);
300                mem::replace(&mut this.const_scope, const_scope)
301            } else {
302                // We don't need to allocate a new scope, since only items matter to us.
303                this.const_scope
304            };
305            // Overwrite the old scope for the block expr, so that every block scope can be found
306            // via the block itthis (important for blocks that only contain items, no expressions).
307            this.scopes.set_scope(expr, this.scope);
308
309            for stmt in statements {
310                match stmt {
311                    Statement::Let { pat, initializer, else_branch, type_ref } => {
312                        this.on_type_opt(*type_ref);
313                        this.on_expr_opt(*initializer);
314                        this.on_expr_opt(*else_branch);
315                        this.scope = this.scopes.new_scope(this.scope);
316                        this.on_pat(*pat);
317                    }
318                    Statement::Expr { expr, has_semi: _ } => this.on_expr(*expr),
319                    Statement::Item(Item::MacroDef(macro_id)) => {
320                        this.scope = this.scopes.new_macro_def_scope(this.scope, macro_id.clone());
321                        this.const_scope =
322                            this.scopes.new_macro_def_scope(this.const_scope, macro_id.clone());
323                    }
324                    Statement::Item(Item::Other) => (),
325                }
326            }
327            this.on_expr_opt(tail);
328
329            this.const_scope = old_const_scope;
330        });
331    }
332}

334impl StoreVisitor for ExprScopeVisitor<'_> {
335    fn on_expr(&mut self, expr: ExprId) {
336        self.scopes.set_scope(expr, self.scope);
337        match &self.store[expr] {
338            Expr::Block { statements, tail, id, label } => {
339                self.visit_block(expr, *id, statements, *tail, *label);
340            }
341            Expr::Unsafe { id, statements, tail } => {
342                self.visit_block(expr, *id, statements, *tail, None);
343            }
344            Expr::Loop { body, label, source: _ } => {
345                let scope = self.scopes.new_labeled_scope(self.scope, *label);
346                self.with_scope(scope, |this| this.on_expr(*body));
347            }
348            Expr::Closure { args, arg_types, ret_type, body, capture_by: _, closure_kind: _ } => {
349                arg_types.iter().for_each(|type_ref| self.on_type_opt(*type_ref));
350                self.on_type_opt(*ret_type);
351                let scope = self.scopes.new_scope(self.scope);
352                self.with_scope(scope, |this| {
353                    this.on_pats(args);
354                    this.on_expr(*body);
355                });
356            }
357            Expr::Match { expr, arms } => {
358                self.on_expr(*expr);
359                for arm in arms.iter() {
360                    let scope = self.scopes.new_scope(self.scope);
361                    self.with_scope(scope, |this| {
362                        this.on_pat(arm.pat);
363                        this.on_expr_opt(arm.guard);
364                        this.on_expr(arm.expr);
365                    });
366                }
367            }
368            &Expr::If { condition, then_branch, else_branch } => {
369                let then_branch_scope = self.scopes.new_scope(self.scope);
370                self.with_scope(then_branch_scope, |this| {
371                    this.on_expr(condition);
372                    this.on_expr(then_branch);
373                });
374                self.on_expr_opt(else_branch);
375            }
376            &Expr::Let { pat, expr } => {
377                self.on_expr(expr);
378                self.scope = self.scopes.new_scope(self.scope);
379                self.on_pat(pat);
380            }
381            _ => self.store.visit_expr_children(expr, self),
382        }
383    }
384
385    fn on_anon_const_expr(&mut self, expr: ExprId) {
386        self.with_scope(self.const_scope, |this| this.on_expr(expr));
387    }
388
389    fn on_pat(&mut self, pat: PatId) {
390        if let Pat::Bind { id, .. } = self.store[pat] {
391            self.scopes.add_bindings(self.store, self.scope, id, self.store.binding_hygiene(id));
392        }
393
394        self.store.visit_pat_children(pat, self);
395    }
396
397    fn on_type(&mut self, ty: TypeRefId) {
398        self.with_scope(self.const_scope, |this| self.store.visit_type_ref_children(ty, this));
399    }
400}

698        let resolved = scopes.resolve_name_in_scope(expr_scope, &name_ref.as_name()).unwrap();
699        let pat_src =
700            source_map.pat_syntax(source_map.patterns_for_binding(resolved.binding())[0]).unwrap();
701
702        let local_name = pat_src.value.syntax_node_ptr().to_node(file.syntax());
703        assert_eq!(local_name.text_range(), expected_name.syntax().text_range());
704    }
